### 2.3 证据 3 · 目录名 ≠ name 字段违例

```bash
$ for d in */; do
    dir=${d%/}
    if [ -f "$d/SKILL.md" ]; then
      n=$(grep '^name:' "$d/SKILL.md" 2>/dev/null | head -1 | awk '{print $2}' | tr -d '"')
      if [ -n "$n" ]; then
        dir_lower=$(echo "$dir" | tr 'A-Z' 'a-z')
        if [ "$dir" != "$n" ] && [ "$dir_lower" != "$n" ]; then
          echo "MISMATCH: dir=$dir name=$n"
        fi
      fi

  done | wc -l
```
